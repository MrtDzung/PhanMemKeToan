## Entity: Account

### C# Domain Entity

```csharp
// Domain/Entities/Account.cs
namespace PhanMemKeToan.Domain.Entities;

public class Account : AuditableEntity  // inherits: TenantId, CreatedAt/By, ModifiedAt/By, IsDeleted
{
    public string AccountNumber { get; set; } = string.Empty;   // max 20, unique per tenant
    public string AccountName { get; set; } = string.Empty;     // max 128, required
    public string? AccountNameEnglish { get; set; }              // max 128, optional
    public Guid? ParentID { get; set; }                          // FK → Account.Id (null = root)
    public int Grade { get; set; }                               // 1=root, computed from parent
    public bool IsParent { get; set; }                           // auto-managed: true when has children

    public AccountCategoryKind AccountCategoryKind { get; set; }
    public bool Inactive { get; set; }
    public bool IsPostableInForeignCurrency { get; set; }

    // Sub-ledger tracking flags (BR-DI03)
    public bool DetailByAccountObject { get; set; }
    public AccountObjectType AccountObjectType { get; set; }
    public bool DetailByBankAccount { get; set; }
    public bool DetailByJob { get; set; }
    public bool DetailByProjectWork { get; set; }
    public bool DetailByOrder { get; set; }
    public bool DetailByContract { get; set; }
    public bool DetailByExpenseItem { get; set; }
    public bool DetailByDepartment { get; set; }
    public bool DetailByListItem { get; set; }
    public bool DetailByPUContract { get; set; }

    public int RowVersion { get; set; }                          // optimistic concurrency (BR-GL04)
    public string? MISACodeID { get; set; }                      // migration reference only

    // Navigation
    public Account? Parent { get; set; }
    public ICollection<Account> Children { get; set; } = [];
}
```

```sql
    "DetailByBankAccount"         boolean         NOT NULL DEFAULT false,
    "DetailByJob"                 boolean         NOT NULL DEFAULT false,
    "DetailByProjectWork"         boolean         NOT NULL DEFAULT false,
    "DetailByOrder"               boolean         NOT NULL DEFAULT false,
    "DetailByContract"            boolean         NOT NULL DEFAULT false,
    "DetailByExpenseItem"         boolean         NOT NULL DEFAULT false,
    "DetailByDepartment"          boolean         NOT NULL DEFAULT false,
    "DetailByListItem"            boolean         NOT NULL DEFAULT false,
    "DetailByPUContract"          boolean         NOT NULL DEFAULT false,

    -- Concurrency + soft delete
    "RowVersion"                  int             NOT NULL DEFAULT 0,
    "IsDeleted"                   boolean         NOT NULL DEFAULT false,
    "MISACodeID"                  varchar(100)    NULL,

    -- Audit (from AuditableEntity)
    "CreatedAt"                   timestamptz     NOT NULL DEFAULT now(),
    "CreatedBy"                   varchar(256)    NULL,
    "ModifiedAt"                  timestamptz     NULL,
    "ModifiedBy"                  varchar(256)    NULL,

    CONSTRAINT "PK_Accounts" PRIMARY KEY ("Id"),
    CONSTRAINT "FK_Accounts_Accounts_ParentID"
        FOREIGN KEY ("ParentID") REFERENCES "Accounts" ("Id") ON DELETE RESTRICT
);
```

## EF Core Configuration

```csharp
// Infrastructure/Persistence/Configurations/AccountConfiguration.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using PhanMemKeToan.Domain.Entities;

namespace PhanMemKeToan.Infrastructure.Persistence.Configurations;

public class AccountConfiguration : IEntityTypeConfiguration<Account>
{
    public void Configure(EntityTypeBuilder<Account> builder)
    {
        builder.ToTable("Accounts");

        builder.HasKey(a => a.Id);

        builder.Property(a => a.AccountNumber)
            .HasMaxLength(20)
            .IsRequired();

        builder.Property(a => a.AccountName)
            .HasMaxLength(128)
            .IsRequired();

        builder.Property(a => a.AccountNameEnglish)
            .HasMaxLength(128);

        builder.Property(a => a.MISACodeID)
            .HasMaxLength(100);

        builder.Property(a => a.AccountCategoryKind)
            .HasConversion<int>();

        builder.Property(a => a.AccountObjectType)
            .HasConversion<int>();

        // Self-referencing tree
        builder.HasOne(a => a.Parent)
            .WithMany(a => a.Children)
            .HasForeignKey(a => a.ParentID)
            .OnDelete(DeleteBehavior.Restrict);

        // Unique account number per tenant (partial — active only)
        builder.HasIndex(a => new { a.TenantId, a.AccountNumber })
            .HasDatabaseName("UIX_Accounts_TenantId_AccountNumber")
            .IsUnique()
            .HasFilter("\"IsDeleted\" = false");

        builder.HasIndex(a => a.ParentID)
            .HasDatabaseName("IX_Accounts_ParentID");

        builder.HasIndex(a => a.TenantId)
            .HasDatabaseName("IX_Accounts_TenantId");
    }
}
```

## Tenant Isolation

`Account` extends `AuditableEntity` which implements `ITenantEntity` (has `TenantId`).  
`ApplicationDbContext` adds global query filter:

```csharp
modelBuilder.Entity<Account>().HasQueryFilter(
    a => a.TenantId == (_tenantContext.TenantId ?? Guid.Empty) && !a.IsDeleted);
```

This ensures all queries automatically filter by the current tenant.

| `GeneralLedger` | References AccountNumber | Read-only check | No FK (cross-module) |
| `AccountDefault` | References AccountNumber | Read-only | No FK (out of scope) |
| `AccountTransfer` | References AccountNumber | Read-only | No FK (out of scope) |

> **Note**: No EF Core FK to `GeneralLedger` — GL entries reference AccountNumber as string, not GUID. The application checks existence via a separate query on delete/code-change.
